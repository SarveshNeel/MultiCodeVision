Create the directory : C:\deps

git clone https://github.com/opencv/opencv.git
cd opencv

git checkout 4.13.0

mkdir build
cd build

cmake .. \
  -G Ninja \
  -DCMAKE_BUILD_TYPE=Release \
  -DBUILD_SHARED_LIBS=OFF \
  -DBUILD_opencv_world=ON \
  -DBUILD_LIST=core,imgproc,imgcodecs,highgui,objdetect \
  -DBUILD_TESTS=OFF \
  -DBUILD_PERF_TESTS=OFF \
  -DBUILD_EXAMPLES=OFF \
  -DBUILD_opencv_apps=OFF \
  -DBUILD_opencv_js=OFF \
  -DBUILD_opencv_java=OFF \
  -DBUILD_opencv_python3=OFF \
  -DBUILD_opencv_python2=OFF \
  -DBUILD_3RDPARTY_LIBS=ON \
  -DWITH_IPP=OFF \
  -DWITH_TBB=OFF \
  -DWITH_OPENMP=ON \
  -DWITH_OPENCL=OFF \
  -DWITH_CUDA=OFF \
  -DWITH_OPENBLAS=OFF \
  -DWITH_LAPACK=OFF \
  -DWITH_EIGEN=OFF \
  -DWITH_FFMPEG=OFF \
  -DWITH_GSTREAMER=OFF \
  -DWITH_MSMF=OFF \
  -DWITH_DSHOW=OFF \
  -DWITH_V4L=OFF \
  -DWITH_1394=OFF \
  -DWITH_OPENEXR=OFF \
  -DWITH_WEBP=OFF \
  -DWITH_TIFF=OFF \
  -DWITH_JASPER=OFF \
  -DWITH_QT=OFF \
  -DWITH_GTK=OFF \
  -DWITH_WIN32UI=ON \
  -DENABLE_AVX2=ON \
  -DENABLE_SSE4_2=ON \
  -DENABLE_FAST_MATH=ON \
  -DCV_ENABLE_INTRINSICS=ON \
  -DCMAKE_CXX_FLAGS="-O3 -march=native -ffast-math" \
  -DCMAKE_INSTALL_PREFIX=C:/deps/opencv

cmake --build . --config Release
cmake --install .



git clone https://github.com/zxing-cpp/zxing-cpp.git
cd zxing-cpp

git fetch --tags
git checkout v3.0.2

  -DZXING_WRITERS=OFF \
  -DZXING_EXAMPLES=OFF \
